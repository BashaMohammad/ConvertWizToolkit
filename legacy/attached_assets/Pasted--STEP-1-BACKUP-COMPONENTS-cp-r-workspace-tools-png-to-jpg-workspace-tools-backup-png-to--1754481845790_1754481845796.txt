# ✅ PNG TO JPG SCRIPT
echo '(function () {
  document.addEventListener("DOMContentLoaded", function () {
    const upload = document.getElementById("upload");
    const canvas = document.createElement("canvas");
    const ctx = canvas.getContext("2d");
    const downloadBtn = document.getElementById("downloadBtn");
    const resetBtn = document.getElementById("resetBtn");

    let jpgData = "";

    upload.addEventListener("change", function () {
      const file = this.files[0];
      if (!file || file.type !== "image/png") {
        alert("Please upload a valid PNG image.");
        return;
      }

      const reader = new FileReader();
      reader.onload = function (e) {
        const img = new Image();
        img.onload = function () {
          canvas.width = img.width;
          canvas.height = img.height;
          ctx.drawImage(img, 0, 0);
          jpgData = canvas.toDataURL("image/jpeg", 0.9);

          document.getElementById("original-img").src = e.target.result;
          document.getElementById("converted-img").src = jpgData;

          document.getElementById("png-size").innerText = "Size: " + (file.size / 1024 / 1024).toFixed(2) + " MB";
          document.getElementById("jpg-size").innerText = "Size: ~" + (jpgData.length * 0.75 / 1024 / 1024).toFixed(2) + " MB";

          document.getElementById("png-name").innerText = "Name: " + file.name;
          document.getElementById("jpg-name").innerText = "Name: " + file.name.replace(/\.png$/i, ".jpg");

          downloadBtn.disabled = false;
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    });

    downloadBtn.addEventListener("click", function () {
      const a = document.createElement("a");
      a.href = jpgData;
      a.download = "converted.jpg";
      a.click();
    });

    resetBtn.addEventListener("click", function () {
      location.reload();
    });
  });
})();' > tools/png-to-jpg/script.js

# ✅ BMI SCRIPT
echo 'function calculateBMI() {
  const h = parseFloat(document.getElementById("height").value) / 100;
  const w = parseFloat(document.getElementById("weight").value);
  const bmiValue = document.getElementById("bmi-value");
  const bmiStatus = document.getElementById("bmi-status");

  if (!h || !w || h <= 0 || w <= 0) {
    bmiValue.innerText = "❗ Enter valid height and weight.";
    bmiStatus.innerText = "";
    return;
  }

  const bmi = w / (h * h);
  let status = "";
  let color = "";

  if (bmi < 18.5) {
    status = "Underweight";
    color = "orange";
  } else if (bmi < 24.9) {
    status = "Normal";
    color = "green";
  } else if (bmi < 29.9) {
    status = "Overweight";
    color = "orange";
  } else {
    status = "Obese";
    color = "red";
  }

  document.getElementById("input-height").innerText = "Height: " + (h * 100).toFixed(0) + " cm";
  document.getElementById("input-weight").innerText = "Weight: " + w.toFixed(1) + " kg";

  bmiValue.innerHTML = `BMI: <strong>${bmi.toFixed(1)}</strong>`;
  bmiStatus.innerHTML = `Status: <span style="color:${color}; font-weight:bold">${status}</span>`;
}' > tools/bmi-calculator/script.js
