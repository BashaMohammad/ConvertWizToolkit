# ✅ STEP 4: ADD LOGIC FOR WIFI SCANNER
# File: tools/wifi-scanner/script.js
echo '(function() {
  document.addEventListener("DOMContentLoaded", function() {
    const output = document.getElementById("wifi-info");

    if ("connection" in navigator) {
      const type = navigator.connection.effectiveType;
      const saveData = navigator.connection.saveData;

      output.innerHTML = `
        <p><strong>Network Type:</strong> ${type.toUpperCase()}</p>
        <p><strong>Data Saver Mode:</strong> ${saveData ? "Enabled" : "Disabled"}</p>
        <p>📡 Browser-reported info. May vary across devices.</p>
      `;
    } else {
      output.innerHTML = `
        <p>❗ Your browser does not support WiFi scanning APIs.</p>
        <p>Try using Chrome on Android for best results.</p>
      `;
    }
  });
})();' > tools/wifi-scanner/script.js

# ✅ STEP 7: ADD LOGIC FOR SIGNAL STRENGTH
# File: tools/signal-scanner/script.js
echo '(function() {
  document.addEventListener("DOMContentLoaded", function() {
    const output = document.getElementById("signal-strength");

    if (navigator.connection && navigator.connection.downlink) {
      const downlink = navigator.connection.downlink;
      const type = navigator.connection.effectiveType;
      const rtt = navigator.connection.rtt;

      output.innerHTML = `
        <p><strong>Connection Type:</strong> ${type.toUpperCase()}</p>
        <p><strong>Estimated Bandwidth:</strong> ${downlink} Mbps</p>
        <p><strong>Estimated Latency:</strong> ${rtt} ms</p>
        <p>📶 Signal info based on browser access. Accuracy may vary.</p>
      `;
    } else {
      output.innerHTML = `
        <p>❗ Signal data not supported in your browser.</p>
        <p>Try using Chrome or Edge on Android.</p>
      `;
    }
  });
})();' > tools/signal-scanner/script.js
