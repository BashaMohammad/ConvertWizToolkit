# script.js
echo 'document.getElementById("convertBtn").addEventListener("click", async () => {
  const input = document.getElementById("pdfFile");
  const status = document.getElementById("statusMsg");
  const file = input.files[0];

  if (!file) {
    status.textContent = "❗ Please upload a PDF file.";
    return;
  }

  status.textContent = "⏳ Reading PDF content...";

  try {
    const buffer = await file.arrayBuffer();
    const pdfDoc = await PDFLib.PDFDocument.load(buffer);
    const pages = pdfDoc.getPages();
    let fullText = "";

    for (const page of pages) {
      const textContent = await page.getTextContent?.(); // fallback
      if (textContent) {
        fullText += textContent.items.map(item => item.str).join(" ") + "\\n\\n";
      } else {
        fullText += "[Text extraction not supported in this mode]\\n";
      }
    }

    const docText = fullText.trim() || "PDF text extraction failed or empty.";

    // Create .docx text file simulation
    const blob = new Blob([docText], { type: "application/vnd.openxmlformats-officedocument.wordprocessingml.document" });
    const downloadLink = document.createElement("a");
    downloadLink.href = URL.createObjectURL(blob);
    downloadLink.download = "converted.docx";
    downloadLink.click();

    status.textContent = "✅ DOCX ready. Download should begin.";
  } catch (err) {
    console.error("PDF to Word error:", err);
    status.textContent = "❌ Failed to convert. Try another file.";
  }
});' > tools/pdf-to-word/script.js
